-- Reprodução da base da base de dados da steam feita para a matéria Modelagem da Informação

CREATE TABLE jogo (
id INT PRIMARY KEY AUTO_INCREMENT,
nome VARCHAR(50) NOT NULL UNIQUE,
class_etaria INT,
categoria VARCHAR(50),
preco FLOAT NOT NULL DEFAULT 0
);
CREATE TABLE usuario (
id INT PRIMARY KEY AUTO_INCREMENT,
username VARCHAR(50) NOT NULL UNIQUE,
email VARCHAR(50) NOT NULL UNIQUE,
nivel INT NOT NULL DEFAULT 0,
saldo FLOAT NOT NULL DEFAULT 0
);
CREATE TABLE compra_jogo (
usuario_id INT,
FOREIGN KEY (usuario_id) REFERENCES usuario(id),
jogo_id INT,
FOREIGN KEY (jogo_id) REFERENCES jogo(id),
preco FLOAT
);
CREATE TABLE desenvolvedora (
id INT PRIMARY KEY AUTO_INCREMENT,
nome VARCHAR(50) NOT NULL
);
CREATE TABLE desenvolve (
desenvolvedora_id INT,
FOREIGN KEY (desenvolvedora_id) REFERENCES desenvolvedora(id),
jogo_id INT,
FOREIGN KEY (jogo_id) REFERENCES jogo(id)
);
CREATE TABLE publica (
desenvolvedora_id INT,
FOREIGN KEY (desenvolvedora_id) REFERENCES desenvolvedora(id),
jogo_id INT,
FOREIGN KEY (jogo_id) REFERENCES jogo(id)
);
CREATE TABLE dlc (
id INT PRIMARY KEY AUTO_INCREMENT,
jogo_base_id INT,
FOREIGN KEY (jogo_base_id) REFERENCES jogo(id),
nome VARCHAR(50) NOT NULL,
preco FLOAT
);
CREATE TABLE compra_dlc (
usuario_id INT,
FOREIGN KEY (usuario_id) REFERENCES usuario(id),
dlc_id INT,
FOREIGN KEY (dlc_id) REFERENCES dlc(id),
preco FLOAT
);
CREATE TABLE achievement (
id INT PRIMARY KEY AUTO_INCREMENT,
jogo_id INT,
FOREIGN KEY (jogo_id) REFERENCES jogo(id),
nome VARCHAR(50)
);
CREATE TABLE conquista (
usuario_id INT,
FOREIGN KEY (usuario_id) REFERENCES usuario(id),
achievement_id INT,
FOREIGN KEY (achievement_id) REFERENCES achievement(id)
);
CREATE TABLE amizade (
usuario_id INT,
FOREIGN KEY (usuario_id) REFERENCES usuario(id),
usuario_id1 INT,
FOREIGN KEY (usuario_id1) REFERENCES usuario(id)
);
CREATE TABLE grupo (
id INT PRIMARY KEY AUTO_INCREMENT,
nome VARCHAR(50)
);
CREATE TABLE membro (
usuario_id INT,
FOREIGN KEY (usuario_id) REFERENCES usuario(id),
grupo_id INT,
FOREIGN KEY (grupo_id) REFERENCES grupo(id),
dono TINYINT
);

-- Relatório: jogos e DLCs dos usuários
SELECT jogo.nome AS Jogos, dlc.nome AS DLC, usuario.username AS Usuario
FROM usuario
INNER JOIN (compra_jogo INNER JOIN jogo ON jogo.id = compra_jogo.jogo_id)
ON compra_jogo.usuario_id = usuario.id
LEFT JOIN compra_dlc ON compra_dlc.usuario_id = usuario.id
LEFT JOIN dlc ON dlc.id = compra_dlc.dlc_id;

-- Relatório: jogos e suas desenvolvedoras
SELECT desenvolvedora.nome AS Empresa, jogo.nome AS Jogo_Desenvolvido
FROM desenvolvedora
INNER JOIN (desenvolve INNER JOIN jogo ON jogo.id = desenvolve.jogo_id)
ON desenvolvedora.id = desenvolve.desenvolvedora_id;

-- Conquistas de um usuário específico
SELECT achievement.nome AS Conquistas
FROM usuario
INNER JOIN (conquista INNER JOIN achievement ON conquista.achievement_id =
achievement.id)
ON conquista.usuario_id = usuario.id
WHERE usuario.username = "Maio Ciranda";

-- Informações completas de um jogo específico
SELECT jogo.nome AS Jogo, jogo.class_etaria AS "Classificação Etária",
jogo.categoria AS Categoria,
dev_dev.nome AS Desenvolvedora, pub_dev.nome AS Publicadora, jogo.preco AS
Preço
FROM jogo
INNER JOIN (publica INNER JOIN desenvolvedora AS pub_dev ON
publica.desenvolvedora_id = pub_dev.id)
ON jogo.id = publica.jogo_id
INNER JOIN (desenvolve INNER JOIN desenvolvedora AS dev_dev ON
desenvolve.desenvolvedora_id = dev_dev.id)
ON jogo.id = desenvolve.jogo_id
WHERE jogo.nome = "Cyberpunk2077";

-- Membros de um grupo
SELECT usuario.username AS Usuário, membro.dono AS Dono
FROM membro
INNER JOIN usuario ON usuario.id = membro.usuario_id
INNER JOIN grupo ON grupo.id = membro.grupo_id;

-- Conquistas de usuários em jogos
SELECT usuario.username AS Usuário, achievement.nome AS Conquista, jogo.nome
AS Jogo
FROM conquista
INNER JOIN usuario ON usuario.id = conquista.usuario_id
INNER JOIN achievement ON achievement.id = conquista.achievement_id
INNER JOIN jogo ON jogo.id = achievement.jogo_id;
